import { Client, Account, TablesDB, Query, ID } from 'node-appwrite'; // =============================== // APPWRITE CONFIG // =============================== const ENDPOINT = process.env.APPWRITE_ENDPOINT; const PROJECT_ID = process.env.APPWRITE_PROJECT_ID; const DATABASE_ID = '6b07af6f000798319ab0'; const API_KEY = process.env.APPWRITE_API_KEY; const TABLES = { rooms: '6b07af9d00191c9f315e', players: '6b07b22000258b36e472', roles: '6b07b38700091cb0fcbe' }; // =============================== // RESPONSE // =============================== function send(res, code, data) { return res.json(data, code); } // =============================== // JWT // =============================== function readJwt(req) { const headers = req.headers || {}; return ( headers['x-appwrite-user-jwt'] || headers['X-Appwrite-User-JWT'] || '' ); } // =============================== // RANDOM ROLE DISTRIBUTION // =============================== function shuffle(array) { const result = [...array]; for (let i = result.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [result[i], result[j]] = [result[j], result[i]]; } return result; } function makeRoleList(count, scenario = 'classic') { const mafiaCount = count >= 12 ? 4 : count >= 9 ? 3 : count >= 6 ? 2 : 1; const roles = []; for (let i = 0; i < mafiaCount; i++) { roles.push('مافیا'); } if (scenario === 'commando') { roles.push('تکاور', 'پزشک'); if (count >= 6) { roles.push('کارآگاه'); } } else if (scenario === 'negotiation') { roles.push('مذاکره‌کننده', 'پزشک'); if (count >= 6) { roles.push('کارآگاه'); } } else { roles.push('پزشک'); if (count >= 5) { roles.push('کارآگاه'); } } while (roles.length < count) { roles.push('شهروند'); } return shuffle(roles.slice(0, count)); } // =============================== // APPWRITE CLIENT // =============================== async function createClients(jwt) { const userClient = new Client() .setEndpoint(ENDPOINT) .setProject(PROJECT_ID) .setJWT(jwt); const account = new Account(userClient); const user = await account.get(); const adminClient = new Client() .setEndpoint(ENDPOINT) .setProject(PROJECT_ID) .setKey(API_KEY); const tables = new TablesDB(adminClient); return { user, tables }; } // =============================== // DATABASE FUNCTIONS // =============================== async function listRows(tables, tableId, queries = []) { const result = await tables.listRows({ databaseId: DATABASE_ID, tableId, queries }); return result.rows || []; } async function updateRoom(tables, roomId, data) { return tables.updateRow({ databaseId: DATABASE_ID, tableId: TABLES.rooms, rowId: roomId, data }); } async function deleteRoles(tables, rows) { for (const row of rows) { await tables.deleteRow({ databaseId: DATABASE_ID, tableId: TABLES.roles, rowId: row.$id }); } } // =============================== // MAIN FUNCTION // =============================== export default async ({ req, res, log, error }) => { try { // CONFIG CHECK if ( !ENDPOINT || !PROJECT_ID || !API_KEY ) { return send(res, 500, { ok: false, error: 'تنظیمات Appwrite کامل نیست.' }); } // AUTH CHECK const jwt = readJwt(req); if (!jwt) { return send(res, 401, { ok: false, error: 'توکن ورود کاربر ارسال نشده است.' }); } const body = req.bodyJson || {}; const action = String(body.action || '').trim(); const roomCode = String( body.roomcode || '' ).trim().toUpperCase(); if (!roomCode) { return send(res, 400, { ok: false, error: 'کد اتاق وارد نشده است.' }); } const { user, tables } = await createClients(jwt); // =============================== // FIND ROOM // =============================== const roomMatches = await listRows( tables, TABLES.rooms, [ Query.equal('roomcode', roomCode), Query.limit(1) ] ); const room = roomMatches[0]; if (!room) { return send(res, 404, { ok: false, error: 'اتاق پیدا نشد.' }); } // =============================== // START GAME // =============================== if (action === 'startGame') { // HOST CHECK if (room.hostId !== user.$id) { return send(res, 403, { ok: false, error: 'فقط میزبان می‌تواند بازی را شروع کند.' }); } // STATUS CHECK if (room.status !== 'waiting') { return send(res, 409, { ok: false, error: 'اتاق در وضعیت انتظار نیست.' }); } // GET PLAYERS const players = await listRows( tables, TABLES.players, [ Query.equal('roomcode', roomCode), Query.limit(100) ] ); if (players.length < 4) { return send(res, 400, { ok: false, error: 'حداقل ۴ بازیکن لازم است.' }); } if ( players.length > Number(room.maxplayers || 12) ) { return send(res, 400, { ok: false, error: 'ظرفیت اتاق تکمیل شده است.' }); } // READY CHECK if ( players.some( player => player.isReady !== true ) ) { return send(res, 400, { ok: false, error: 'همه بازیکنان باید آماده باشند.' }); } // DUPLICATE CHECK const userIds = players.map( player => player.UserId ); if ( userIds.some(id => !id) || new Set(userIds).size !== players.length ) { return send(res, 409, { ok: false, error: 'بازیکن تکراری یا شناسه نامعتبر وجود دارد.' }); } // SET STARTING STATUS await updateRoom( tables, room.$id, { status: 'starting' } ); const createdRoleIds = []; try { // DELETE OLD ROLES const oldRoles = await listRows( tables, TABLES.roles, [ Query.equal('roomcode', roomCode), Query.limit(100) ] ); await deleteRoles(tables, oldRoles); // CREATE NEW ROLES const assignedRoles = makeRoleList( players.length, room.scenario || 'classic' ); for ( let i = 0; i < players.length; i++ ) { const roleId = ID.unique(); await tables.createRow({ databaseId: DATABASE_ID, tableId: TABLES.roles, rowId: roleId, data: { roomcode: roomCode, userId: players[i].UserId, role: assignedRoles[i], isRevealed: false } }); createdRoleIds.push(roleId); } // ACTIVATE GAME await updateRoom( tables, room.$id, { status: 'active', currentphase: 'night', phaseStartedAt: new Date().toISOString(), winner: '' } ); log( `Game started: ${roomCode}` ); return send(res, 200, { ok: true, message: 'بازی با موفقیت شروع شد.', playerCount: players.length }); } catch (startError) { // REMOVE PARTIAL ROLE ASSIGNMENTS for (const roleId of createdRoleIds) { try { await tables.deleteRow({ databaseId: DATABASE_ID, tableId: TABLES.roles, rowId: roleId }); } catch (cleanupError) { error( String(cleanupError) ); } } // RESTORE WAITING STATUS try { await updateRoom( tables, room.$id, { status: 'waiting' } ); } catch (rollbackError) { error( String(rollbackError) ); } throw startError; } } // =============================== // GET MY ROLE // =============================== if (action === 'myRole') { const players = await listRows( tables, TABLES.players, [ Query.equal('roomcode', roomCode), Query.equal('UserId', user.$id), Query.limit(1) ] ); if (!players.length) { return send(res, 403, { ok: false, error: 'شما عضو این اتاق نیستید.' }); } const roles = await listRows( tables, TABLES.roles, [ Query.equal('roomcode', roomCode), Query.equal('userId', user.$id), Query.limit(1) ] ); if (!roles.length) { return send(res, 404, { ok: false, error: 'نقش شما هنوز تعیین نشده است.' }); } return send(res, 200, { ok: true, role: roles[0].role }); } // =============================== // MODERATOR ROLES // =============================== if (action === 'moderatorRoles') { const authorized = room.hostId === user.$id || room.assistantModeratorId === user.$id; if (!authorized) { return send(res, 403, { ok: false, error: 'شما اجازه مشاهده نقش‌ها را ندارید.' }); } const roles = await listRows( tables, TABLES.roles, [ Query.equal('roomcode', roomCode), Query.limit(100) ] ); return send(res, 200, { ok: true, roles: roles.map(row => ({ userId: row.userId, role: row.role })) }); } // =============================== // INVALID ACTION // =============================== return send(res, 400, { ok: false, error: 'عملیات نامعتبر است.' }); } catch (err) { error( err?.stack || String(err) ); return send(res, 500, { ok: false, error: 'خطای داخلی تابع.', details: err?.message || String(err) }); } };
